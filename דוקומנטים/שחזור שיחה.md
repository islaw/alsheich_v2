```text
You are an expert Israeli legal-tech AI assistant. Your task is to accurately transcribe and structure the content of an image of a court protocol page into a structured JSON format, COMPLETELY IGNORING the header and footer sections.

**Instructions:**
1.  **Output Format:** Provide a single, valid JSON object. Do not include any text, explanations, or code formatting (like ```json) before or after the JSON object. The structure must be exactly as follows. **Note that there is NO "header_content" field:**

    ```json
    {
      "page_metadata": {
        "visual_page_number": "string_or_null"
      },
      "protocol_body": [
        {
          "line_number": "string_or_null",
          "speaker": "string_or_null",
          "text": "string"
        },
        ...
      ]
    }
    ```

2.  **Header & Footer Zones (CRITICAL: IGNORE):**
    *   **Header:** The text at the very top of the page (e.g., "בית המשפט המחוזי בירושלים", case numbers). **You MUST completely ignore this section.**
    *   **Footer:** The text at the very bottom of the page, including page numbers and signatures. **You MUST completely ignore this section.**

3.  **Main Content (protocol_body):**
    * **Fundamental Rule:** All textual content to be extracted for the "protocol_body" is aligned with a line number, usually on the left side of the main text body. If a piece of text on the page does not have a corresponding line number, it is not part of the main protocol body and should be ignored.
    * **Speaker and Speaking Turns:** The protocol is a record of a multi-participant conversation, structured as a series of "speaking turns."
        * A new speaking turn begins whenever a speaker identifier appears. This can be a name, a title (e.g., "עו\\"ד כהן:", "ת:", "ש:"), or an abbreviation (e.g., "ש.", "ת.").
        * Extract this speaker identifier **EXACTLY** as it is written, but **without the trailing colon**. Place it in the `speaker` field.
        * If a line is not spoken by a specific person (e.g., "הפרוטוקול הוקלט", "החלטה"), the `speaker` field should be `null`.
    * **Content Aggregation:**
        * A speaker's turn consists of the speaker identifier AND the dialogue that immediately follows it. The text of the dialogue may start on the same physical line as the speaker, or it may start on the subsequent line(s). You must collect all consecutive lines of text that belong to that speaker's turn and combine them into a single `text` entry.
        * The `line_number` for the entire speaking turn should be the number of the *first* line where the speaker was identified.
        * The end of a sentence (e.g., with a period or question mark) **does NOT** signal the end of a turn. A speaking turn **ONLY** ends when a new, different speaker identifier appears on a subsequent line.
    * **Text Accuracy:** The `text` field must contain the exact, verbatim text from the image, including all punctuation. Do not add, remove, or change any words.

import base64
import google.generativeai as genai
import difflib
import re
from datetime import datetime

from dotenv import load_dotenv
load_dotenv()
API_KEY = os.getenv("GEMINI_API_KEY")

GROUND_TRUTH_JSON_DIR = 'downloaded_json_parts'

def load_images_for_chunk(chunk_index, pages_per_chunk):
    """Loads and encodes all images for a given chunk index."""
    encoded_images = []
    start_page = (chunk_index * pages_per_chunk) + 1
    end_page = start_page + pages_per_chunk
    
    print(f"טוען תמונות עבור עמודים {start_page} עד {end_page - 1}...")
    
    for page_num in range(start_page, end_page):
        path = os.path.join(IMAGE_INPUT_DIR, f"page_{page_num}.jpg")
        if not os.path.exists(path):
            print(f"שגיאה: התמונה לא נמצאה בנתיב {path}. עוצר.")
            return None
        with open(path, "rb") as image_file:
            encoded_data = base64.b64encode(image_file.read()).decode('utf-8')
            encoded_images.append({"mime_type": "image/jpeg", "data": encoded_data})
    return encoded_images

def get_structured_list_from_gt(file_path):
    """
    Extracts and structures the protocol body from the raw Document AI JSON file.
    This function is critical for creating a clean, comparable ground truth.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            data = json.load(f)
        
        full_text = data.get('text', '')
        lines = full_text.split('\n')
        
        structured_body = []
        # Regex to find a line number at the start of a line
        line_num_pattern = re.compile(r'^\s*(\d+)\s*(.*)')
        
        for line in lines:
            match = line_num_pattern.match(line)
            if match:
                line_number = match.group(1)
                content = match.group(2).strip()
                
                # Simple speaker detection (can be improved)
                speaker_match = re.match(r'([^:]+):\s*(.*)', content)
                if speaker_match:
                    speaker = speaker_match.group(1).strip()
                    text = speaker_match.group(2).strip()
                else:
                    speaker = None
                    text = content
                
                if text: # Only add lines with actual text content
                    structured_body.append({
                        "line_number": line_number,
                        "speaker": speaker,
                        "text": text
                    })
        return structured_body
    except Exception as e:
        print(f"שגיאה בעיבוד קובץ האמת המוחלטת {file_path}: {e}")
        return []

def load_error_traps(path):
    """Loads the error traps JSON file."""
    if not os.path.exists(path):
        print("קובץ מלכודות שגיאה לא נמצא. ממשיך בלעדיו.")
        return {}
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except json.JSONDecodeError:
        print(f"אזהרה: לא ניתן היה לפענח את ה-JSON מהקובץ {path}. ממשיך ללא מלכודות.")
        return {}

def assemble_prompt(base_prompt, traps):
    """Appends error traps to the base prompt."""
    if not traps:
        return base_prompt
    
    traps_text = "\n\n**כללי תמלול נוספים (מלכודות שגיאה):**\n"
    traps_text += "יש ליישם את תיקוני התמלול הספציפיים הבאים. זוהי הוראה קריטית.\n"
    for error, correction in traps.items():
        traps_text += f"- אם אתה מזהה את הטקסט \"{error}\", עליך לתמלל אותו כ-\"{correction}\".\n"
        
    return base_prompt + traps_text

def get_model_response(prompt, images):
    """Sends the request to the Generative AI model."""
    generation_config = {"temperature": 0.0, "response_mime_type": "application/json"}
    model = genai.GenerativeModel(model_name=MODEL_NAME, generation_config=generation_config)
    content_parts = [prompt] + images
    try:
        print(f"שולח בקשה אל {MODEL_NAME}...")
        response = model.generate_content(content_parts)
        return response.text
    except Exception as e:
        print(f"אירעה שגיאה במהלך קריאת ה-API: {e}")
        return f'{{"error": "API call failed: {str(e)}"}}'

def parse_model_output(raw_output):
    """Parses the model's raw string output into a Python dictionary."""
    # Clean potential markdown formatting
    cleaned_output = re.sub(r'```json\s*|\s*```', '', raw_output).strip()
    try:
        return json.loads(cleaned_output)
    except json.JSONDecodeError as e:
        print(f"שגיאה בפענוח פלט ה-JSON של המודל: {e}")
        print("--- פלט גולמי מהמודל ---")
        print(raw_output)
        print("------------------------")
        return {"error": "Failed to parse JSON", "raw_output": raw_output}

def format_for_diff(structured_list):
    """Formats a structured list into a list of human-readable strings for difflib."""
    if not isinstance(structured_list, list):
        return ["שגיאה: הקלט אינו רשימה."]
        
    formatted_lines = []
    for item in structured_list:
        line_num = item.get('line_number', 'N/A')
        speaker = item.get('speaker', 'None')
        text = item.get('text', '')
        formatted_lines.append(f"[{line_num}] {speaker}: {text}")
    return formatted_lines

def create_diff_report(gt_lines, model_lines, file_path):
    """Generates and saves an HTML diff report."""
    diff = difflib.HtmlDiff(wrapcolumn=80).make_file(
        fromlines=gt_lines, tolines=model_lines,
        fromdesc='אמת מוחלטת (מעובד)', todesc='פלט המודל (מובנה)'
    )
    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(diff)
    print(f"דוח השוואה נוצר בנתיב: {file_path}")

def main():
    print(f"--- מתחיל הרצה של רתמת בדיקות v2.0, סבב מספר {ROUND_NUMBER} ---")
    
    # 1. Load Ground Truth
    gt_path = os.path.join(GROUND_TRUTH_JSON_DIR, f"full_ground_truth_protocol-{CHUNK_INDEX_TO_TEST}.json")
    ground_truth_structured = get_structured_list_from_gt(gt_path)
    if not ground_truth_structured:
        print("לא ניתן היה לקבל אמת מוחלטת מובנית. עוצר.")
        return

    # 2. Load Prompt and Error Traps
    try:
        with open(PROMPT_FILE_PATH, 'r', encoding='utf-8') as f:
            base_prompt = f.read()
    except FileNotFoundError:
        print(f"קובץ הפרומפט לא נמצא בנתיב {PROMPT_FILE_PATH}. עוצר.")
        return
        
    error_traps = load_error_traps(ERROR_TRAPS_PATH)
    final_prompt = assemble_prompt(base_prompt, error_traps)

    # 3. Load Images
    images = load_images_for_chunk(CHUNK_INDEX_TO_TEST, PAGES_PER_CHUNK)
    if not images:
        return

    # 4. Get Model Response
    model_raw_output = get_model_response(final_prompt, images)
    
    # 5. Save Raw Output & Parse
    prompt_name = os.path.splitext(os.path.basename(PROMPT_FILE_PATH))[0]
    base_filename = f"round{ROUND_NUMBER}_chunk{CHUNK_INDEX_TO_TEST}_model_{MODEL_NAME}_prompt_{prompt_name}"
    
    json_output_path = os.path.join(JSON_OUTPUTS_DIR, f"{base_filename}.json")
    with open(json_output_path, 'w', encoding='utf-8') as f:
        f.write(model_raw_output)

    model_structured_output = parse_model_output(model_raw_output)

    gt_for_diff = format_for_diff(ground_truth_structured)
    model_body_for_diff = format_for_diff(model_structured_output.get('protocol_body', []))
    
    report_path = os.path.join(REPORTS_DIR, f"{base_filename}_final_report.html")
    create_diff_report(gt_for_diff, model_body_for_diff, report_path)
    
    print(f"--- הרצה מספר {ROUND_NUMBER} הסתיימה. הדוח נמצא ב: {report_path} ---")

if __name__ == "__main__":
    main()

        print(f"אזהרה: לא ניתן היה לפענח את ה-JSON מהקובץ {path}. ממשיך ללא מלכודות.")
        return {}

def assemble_prompt(base_prompt, traps):
    """Appends error traps to the base prompt."""
    if not traps:
        return base_prompt
    
    traps_text = "\n\n**כללי תמלול נוספים (מלכודות שגיאה):**\n"
    traps_text += "יש ליישם את תיקוני התמלול הספציפיים הבאים. זוהי הוראה קריטית.\n"
    for error, correction in traps.items():
        traps_text += f"- אם אתה מזהה את הטקסט \"{error}\", עליך לתמלל אותו כ-\"{correction}\".\n"
        
    return base_prompt + traps_text

def get_model_response(prompt, images):
    """Sends the request to the Generative AI model."""
    generation_config = {"temperature": 0.0, "response_mime_type": "application/json"}
    model = genai.GenerativeModel(model_name=MODEL_NAME, generation_config=generation_config)
    content_parts = [prompt] + images
    try:
        print(f"שולח בקשה אל {MODEL_NAME}...")
        response = model.generate_content(content_parts)
        return response.text
    except Exception as e:
        print(f"אירעה שגיאה במהלך קריאת ה-API: {e}")
        return f'{{"error": "API call failed: {str(e)}"}}'

def parse_model_output(raw_output):
    """Parses the model's raw string output into a Python dictionary."""
    # Clean potential markdown formatting
    cleaned_output = re.sub(r'```json\s*|\s*```', '', raw_output).strip()
    try:
        return json.loads(cleaned_output)
    except json.JSONDecodeError as e:
        print(f"שגיאה בפענוח פלט ה-JSON של המודל: {e}")
        print("--- פלט גולמי מהמודל ---")
        print(raw_output)
        print("------------------------")
        return {"error": "Failed to parse JSON", "raw_output": raw_output}

def format_for_diff(structured_list):
    """Formats a structured list into a list of human-readable strings for difflib."""
    if not isinstance(structured_list, list):
        return ["שגיאה: הקלט אינו רשימה."]
        
    formatted_lines = []
    for item in structured_list:
        line_num = item.get('line_number', 'N/A')
        speaker = item.get('speaker', 'None')
        text = item.get('text', '')
        formatted_lines.append(f"[{line_num}] {speaker}: {text}")
    return formatted_lines

def create_diff_report(gt_lines, model_lines, file_path):
    """Generates and saves an HTML diff report."""
    diff = difflib.HtmlDiff(wrapcolumn=80).make_file(
        fromlines=gt_lines, tolines=model_lines,
        fromdesc='אמת מוחלטת (מעובד)', todesc='פלט המודל (מובנה)'
    )
    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(diff)
    print(f"דוח השוואה נוצר בנתיב: {file_path}")
